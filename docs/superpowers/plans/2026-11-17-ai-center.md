# AI Center Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Windows tray app (Node + PowerShell WinForms helper) exposing a localhost REST API to play category sounds, open URLs after a Yes/No popup, and open the SkilledPlus4U MCP console from the tray menu.

**Architecture:** Node holds all state and decisions (config, sound toggle, confirm queue, console URL) and serves HTTP on 127.0.0.1. A long-running PowerShell helper renders the tray icon, menu and popup and plays sounds; the two talk in JSON lines over the helper's stdin/stdout. Everything except the helper and `main.js` wiring is unit-tested with `node:test` and runs on Linux/WSL.

**Tech Stack:** Node ≥ 20 (ESM, `node:http`, `node:test`), no npm dependencies; Windows PowerShell 5.1 + WinForms (+ inline C# via `Add-Type`).

**Spec:** `docs/superpowers/specs/2026-11-17-ai-center-design.md`

## Global Constraints

- No npm dependencies (`dependencies` and `devDependencies` stay empty).
- `"type": "module"`, `"engines": { "node": ">=20" }`.
- Server binds only to `127.0.0.1`.
- Default config: `port` 7399, `soundsDir` `%USERPROFILE%\.claude-sound`, `confirmSoundDir` `%USERPROFILE%\.claude-sound\confirm`, `confirmTimeoutSec` 30.
- Sound categories: `fail`, `done`, `ask`, `waiting60s`.
- `config.json` and `state.json` live in the app root (next to `package.json`) and are git-ignored.
- `/sound` always answers `200`.
- Confirm queue max 10 (including the shown popup); identical URLs are NOT merged.
- `/console-url` accepts only `http(s)` on `127.0.0.1`, `localhost`, `[::1]`.
- `.ps1` files are saved as UTF-8 **with BOM** (Windows PowerShell 5.1 reads BOM-less files as ANSI and breaks Czech text).
- Helper never writes anything to stdout except protocol JSON lines.
- Messages Node → helper are ASCII-only (non-ASCII escaped as `\uXXXX`).

const MAX_URL_LENGTH = 2048;
const LOOPBACK_HOSTS = new Set(["127.0.0.1", "localhost", "[::1]"]);

/** An absolute http(s) URL, or null for anything else. */
export function parseHttpUrl(value) {
  if (typeof value !== "string" || value.length === 0 || value.length > MAX_URL_LENGTH) return null;
  let url;
  try {
    url = new URL(value);
  } catch {
    return null;
  }
  return url.protocol === "http:" || url.protocol === "https:" ? url : null;
}

/** True for an http(s) URL pointing at this machine's loopback interface. */
export function isLoopbackHttpUrl(value) {
  const url = parseHttpUrl(value);
  return url !== null && LOOPBACK_HOSTS.has(url.hostname);
}

import { existsSync, readFileSync, writeFileSync } from "node:fs";
import { CONFIG_FILE } from "./paths.js";

export const DEFAULT_CONFIG = Object.freeze({
  port: 7399,
  soundsDir: "%USERPROFILE%\\.claude-sound",
  confirmSoundDir: "%USERPROFILE%\\.claude-sound\\confirm",
  confirmTimeoutSec: 30
});

/** Expands Windows-style %NAME% variables; unknown ones stay as written. */
export function expandEnv(value, env = process.env) {
  return value.replace(/%([^%]+)%/g, (match, name) => env[name] ?? match);
}

function check(ok, key, what) {
  if (!ok) throw new Error(`config.json: "${key}" must be ${what}`);
}

const isPath = value => typeof value === "string" && value.trim().length > 0;

export function loadConfig(file = CONFIG_FILE, env = process.env) {
  let raw = {};
  if (!existsSync(file)) {
    writeFileSync(file, JSON.stringify(DEFAULT_CONFIG, null, 2) + "\n", "utf-8");
  } else {
    const text = readFileSync(file, "utf-8").replace(/^\uFEFF/, "");
    try {
      raw = JSON.parse(text);
    } catch (error) {
      throw new Error(`Invalid JSON in ${file}: ${error.message}`);
    }
  }
  if (raw === null || typeof raw !== "object" || Array.isArray(raw)) {
    throw new Error(`config.json must contain a JSON object (${file})`);
  }

  const merged = {};
  for (const key of Object.keys(DEFAULT_CONFIG)) {
    merged[key] = Object.hasOwn(raw, key) ? raw[key] : DEFAULT_CONFIG[key];
  }
  check(Number.isInteger(merged.port) && merged.port >= 1 && merged.port <= 65535, "port", "an integer 1-65535");
  check(isPath(merged.soundsDir), "soundsDir", "a non-empty path");
  check(isPath(merged.confirmSoundDir), "confirmSoundDir", "a non-empty path");
  check(Number.isInteger(merged.confirmTimeoutSec) && merged.confirmTimeoutSec >= 5 && merged.confirmTimeoutSec <= 600,
    "confirmTimeoutSec", "an integer 5-600");

  return Object.freeze({
    port: merged.port,
    soundsDir: expandEnv(merged.soundsDir, env),
    confirmSoundDir: expandEnv(merged.confirmSoundDir, env),
    confirmTimeoutSec: merged.confirmTimeoutSec
  });
}

import { readFileSync, renameSync, writeFileSync } from "node:fs";

export function loadState(file) {
  try {
    const data = JSON.parse(readFileSync(file, "utf-8").replace(/^\uFEFF/, ""));
    return { soundEnabled: data?.soundEnabled !== false };
  } catch {
    return { soundEnabled: true };
  }
}

export function saveState(file, state) {
  const temp = `${file}.tmp`;
  writeFileSync(temp, JSON.stringify({ soundEnabled: state.soundEnabled }, null, 2) + "\n", "utf-8");
  renameSync(temp, file);
}

import { join } from "node:path";

export const SOUND_CATEGORIES = Object.freeze(["fail", "done", "ask", "waiting60s"]);

export function pickRandomWav(dir, random = Math.random) {
  let names;
  try {
    names = readdirSync(dir, { withFileTypes: true })
      .filter(entry => entry.isFile() && entry.name.toLowerCase().endsWith(".wav"))
      .map(entry => entry.name)
      .sort();
  } catch {
    return null;
  }
  if (names.length === 0) return null;
  return join(dir, names[Math.floor(random() * names.length)]);
}

function defaultSetTimer(fn, ms) {
  const timer = setTimeout(fn, ms);
  timer.unref?.();
  return timer;
}

/**
 * FIFO of URLs waiting for the user's Yes/No. Exactly one is shown at a time;
 * the rest wait. Talks to the UI only through the three callbacks.
 */
export class ConfirmQueue {
  #pending = [];
  #current = null;
  #nextId = 1;

  constructor({ max, timeoutSec, graceMs = 10000, onShow, onWaiting, onCancel,
    setTimer = defaultSetTimer, clearTimer = clearTimeout }) {
    this.max = max;
    this.timeoutSec = timeoutSec;
    this.graceMs = graceMs;
    this.onShow = onShow;
    this.onWaiting = onWaiting;
    this.onCancel = onCancel;
    this.setTimer = setTimer;
    this.clearTimer = clearTimer;
  }

  get size() {
    return this.#pending.length + (this.#current ? 1 : 0);
  }

  enqueue(url) {
    if (this.size >= this.max) return null;
    let resolve;
    const promise = new Promise(r => { resolve = r; });
    const item = { id: this.#nextId++, url, resolve, timer: null };
    this.#pending.push(item);
    if (this.#current) this.onWaiting(this.#current.id, this.#pending.length);
    else this.#showNext();
    return { id: item.id, promise };
  }

  resolve(id, result) {
    if (this.#current?.id !== id) return false;
    this.#finish(result);
    return true;
  }

  cancel(id) {
    if (this.#current?.id === id) {
      this.onCancel(id);
      this.#finish("cancelled");
      return true;
    }
    const index = this.#pending.findIndex(item => item.id === id);
    if (index < 0) return false;
    const [item] = this.#pending.splice(index, 1);
    item.resolve("cancelled");
    if (this.#current) this.onWaiting(this.#current.id, this.#pending.length);
    return true;
  }

  failAll(reason) {
    const items = this.#current ? [this.#current, ...this.#pending] : [...this.#pending];
    if (this.#current) this.clearTimer(this.#current.timer);
    this.#current = null;
    this.#pending = [];
    for (const item of items) item.resolve(reason);
  }

  #finish(result) {
    const item = this.#current;
    this.clearTimer(item.timer);
    this.#current = null;
    item.resolve(result);
    this.#showNext();
  }

  #showNext() {
    const item = this.#pending.shift();
    if (!item) return;
    this.#current = item;
    item.timer = this.setTimer(() => {
      this.onCancel(item.id);
      this.#finish("timeout");
    }, this.timeoutSec * 1000 + this.graceMs);
    this.onShow({ id: item.id, url: item.url }, this.#pending.length);
  }
}

import { spawn } from "node:child_process";
import { EventEmitter } from "node:events";
import { createInterface } from "node:readline";

/** One protocol line, ASCII only so the console code page never matters. */
export function encodeMessage(message) {
  return JSON.stringify(message).replace(/[\u007f-\uffff]/g,
    ch => "\\u" + ch.charCodeAt(0).toString(16).padStart(4, "0")) + "\n";
}

/**
 * The PowerShell tray helper as a child process speaking JSON lines.
 * Restarts it with exponential backoff when it dies unexpectedly.
 */
export class Helper extends EventEmitter {
  #child = null;
  #stopping = false;
  #attempts = 0;
  #restartTimer = null;

  constructor({ command, args, spawnFn = spawn, baseDelayMs = 5000, maxDelayMs = 60000, maxRestarts = 5 }) {
    super();
    Object.assign(this, { command, args, spawnFn, baseDelayMs, maxDelayMs, maxRestarts });
  }

  start() {
    const child = this.spawnFn(this.command, this.args, { stdio: ["pipe", "pipe", "pipe"], windowsHide: true });
    this.#child = child;
    createInterface({ input: child.stdout }).on("line", line => this.#onLine(line));
    createInterface({ input: child.stderr }).on("line", line => {
      if (line.trim()) this.emit("message", { event: "log", message: `stderr: ${line}` });
    });
    child.stdin.on("error", () => { /* EPIPE while the helper is dying */ });
    child.on("error", error => {
      this.emit("message", { event: "log", message: `helper failed: ${error.message}` });
      this.#onExit(child, null);
    });
    child.on("exit", code => this.#onExit(child, code));
  }

  send(message) {
    const child = this.#child;
    if (!child || !child.stdin.writable) return false;
    child.stdin.write(encodeMessage(message));
    return true;
  }

  stop() {
    this.#stopping = true;
    clearTimeout(this.#restartTimer);
    const child = this.#child;
    if (!child) return;
    child.stdin.end();
    const killTimer = setTimeout(() => child.kill(), 3000);
    killTimer.unref?.();
    child.once("exit", () => clearTimeout(killTimer));
  }

  #onLine(raw) {
    const line = raw.replace(/^\uFEFF/, "").trim();
    if (!line) return;
    let message;
    try {
      message = JSON.parse(line);
    } catch {
      this.emit("message", { event: "log", message: `unparsable helper output: ${line}` });
      return;
    }
    if (message === null || typeof message !== "object") return;
    if (message.event === "ready") this.#attempts = 0;
    this.emit("message", message);
  }

  #onExit(child, code) {
    if (this.#child !== child) return;
    this.#child = null;
    if (this.#stopping) return;
    this.emit("down", code);
    if (this.#attempts >= this.maxRestarts) {
      this.emit("gave-up");
      return;
    }
    const delay = Math.min(this.maxDelayMs, this.baseDelayMs * 2 ** this.#attempts);
    this.#attempts += 1;
    this.#restartTimer = setTimeout(() => {
      this.#restartTimer = null;
      if (!this.#stopping) this.start();
    }, delay);
  }
}
```

import { basename, join } from "node:path";
import { ConfirmQueue } from "./confirmQueue.js";
import { SOUND_CATEGORIES, pickRandomWav } from "./sounds.js";

const CONFIRM_RESULTS = new Set(["yes", "no", "timeout"]);

/** The app's state and decisions; the HTTP server and the helper only talk to this. */
export function createApp({ config, helper, version, loadState, saveState, onExit,
  logger = console, pickWav = pickRandomWav, maxPending = 10 }) {
  let soundEnabled = loadState().soundEnabled;
  let consoleUrl = null;

  function playFrom(dir) {
    if (!soundEnabled) return { played: false, reason: "muted" };
    const file = pickWav(dir);
    if (!file) return { played: false, reason: "no-sounds" };
    helper.send({ cmd: "play", path: file });
    return { played: true, file: basename(file) };
  }

  const queue = new ConfirmQueue({
    max: maxPending,
    timeoutSec: config.confirmTimeoutSec,
    onShow: (item, waiting) => {
      playFrom(config.confirmSoundDir);
      helper.send({ cmd: "confirm", id: item.id, url: item.url, timeoutSec: config.confirmTimeoutSec, waiting });
    },
    onWaiting: (id, count) => helper.send({ cmd: "waiting", id, count }),
    onCancel: id => helper.send({ cmd: "cancel", id })
  });

  return {
    playCategory(category) {
      if (!SOUND_CATEGORIES.includes(category)) return { played: false, reason: "unknown-category" };
      return playFrom(join(config.soundsDir, category));
    },

    requestOpen(url) {
      const ticket = queue.enqueue(url);
      if (!ticket) return null;
      return {
        id: ticket.id,
        promise: ticket.promise.then(result => {
          if (result === "yes") helper.send({ cmd: "open", url });
          return result;
        })
      };
    },

    cancelOpen(id) {
      queue.cancel(id);
    },

    setConsoleUrl(url) {
      consoleUrl = url;
      helper.send({ cmd: "consoleUrl", url });
      return consoleUrl;
    },

    health() {
      return {
        ok: true, version, port: config.port, soundEnabled,
        consoleUrl: consoleUrl !== null, pendingConfirms: queue.size
      };
    },

    handleHelperMessage(message) {
      switch (message.event) {
        case "ready":
          helper.send({ cmd: "init", header: `AI Center · port ${config.port}`, soundEnabled });
          helper.send({ cmd: "consoleUrl", url: consoleUrl });
          break;
        case "confirm":
          if (CONFIRM_RESULTS.has(message.result)) queue.resolve(message.id, message.result);
          break;
        case "toggleSound":
          soundEnabled = !soundEnabled;
          try {
            saveState({ soundEnabled });
          } catch (error) {
            logger.error(`Cannot save state: ${error.message}`);
          }
          helper.send({ cmd: "soundEnabled", value: soundEnabled });
          break;
        case "exit":
          onExit();
          break;
        case "log":
          logger.log(`[helper] ${message.message}`);
          break;
        default:
          break;
      }
    },

    handleHelperDown() {
      queue.failAll("helper-restarted");
    },

    shutdown() {
      queue.failAll("shutdown");
    }
  };
}

import http from "node:http";
import { isLoopbackHttpUrl, parseHttpUrl } from "./urls.js";

export const MAX_BODY_BYTES = 16 * 1024;

const ROUTES = {
  "/health": ["GET"],
  "/sound": ["POST"],
  "/open": ["POST"],
  "/console-url": ["POST", "DELETE"]
};

function sendJson(res, status, body, headers = {}) {
  if (res.destroyed) return;
  const data = JSON.stringify(body);
  res.writeHead(status, {
    "Content-Type": "application/json; charset=utf-8",
    "Content-Length": Buffer.byteLength(data),
    ...headers
  });
  res.end(data);
}

/** The parsed JSON object, or undefined for anything unusable (too big, broken, not an object). */
async function readJson(req) {
  const chunks = [];
  let size = 0;
  for await (const chunk of req) {
    size += chunk.length;
    if (size <= MAX_BODY_BYTES) chunks.push(chunk);
  }
  if (size > MAX_BODY_BYTES) return undefined;
  try {
    const value = JSON.parse(Buffer.concat(chunks).toString("utf-8"));
    return value !== null && typeof value === "object" && !Array.isArray(value) ? value : undefined;
  } catch {
    return undefined;
  }
}

async function handleOpen(app, body, res) {
  const url = body ? parseHttpUrl(body.url) : null;
  if (!url) return sendJson(res, 400, { error: "url must be an absolute http(s) URL" });
  const ticket = app.requestOpen(url.href);
  if (!ticket) return sendJson(res, 429, { error: "too many pending confirmations" });
  res.on("close", () => {
    if (!res.writableFinished) app.cancelOpen(ticket.id);
  });
  const result = await ticket.promise;
  if (result === "cancelled") return;
  sendJson(res, 200, result === "yes"
    ? { opened: true }
    : { opened: false, reason: result === "no" ? "rejected" : result });
}

async function handle(app, req, res) {
  const path = new URL(req.url, "http://localhost").pathname;
  const allowed = ROUTES[path];
  if (!allowed) return sendJson(res, 404, { error: "not found" });
  if (!allowed.includes(req.method)) {
    return sendJson(res, 405, { error: "method not allowed" }, { Allow: allowed.join(", ") });
  }
  if (path === "/health") return sendJson(res, 200, app.health());
  if (path === "/console-url" && req.method === "DELETE") {
    return sendJson(res, 200, { consoleUrl: app.setConsoleUrl(null) });
  }

  const body = await readJson(req);
  if (path === "/sound") {
    return sendJson(res, 200, body ? app.playCategory(body.category) : { played: false, reason: "bad-request" });
  }
  if (path === "/console-url") {
    if (!body || !isLoopbackHttpUrl(body.url)) {
      return sendJson(res, 400, { error: "url must be an http(s) URL on 127.0.0.1, localhost or [::1]" });
    }
    return sendJson(res, 200, { consoleUrl: app.setConsoleUrl(parseHttpUrl(body.url).href) });
  }
  return handleOpen(app, body, res);
}

export function createServer(app, { logger = console } = {}) {
  return http.createServer((req, res) => {
    handle(app, req, res).catch(error => {
      logger.error(`Request failed: ${error.stack ?? error}`);
      if (!res.headersSent) sendJson(res, 500, { error: error.message });
      else res.destroy();
    });
  });
}
```

# ---------------------------------------------------------------- protocol

function Invoke-HelperCommand($Msg) {
  switch ([string]$Msg.cmd) {
    'init' {
      $script:header = [string]$Msg.header
      $script:soundEnabled = [bool]$Msg.soundEnabled
      Update-Tray
    }
    'soundEnabled' { $script:soundEnabled = [bool]$Msg.value; Update-Tray }
    'consoleUrl' {
      $script:consoleUrl = if ($Msg.url) { [string]$Msg.url } else { $null }
      Update-Tray
    }
    'play' { Start-Sound ([string]$Msg.path) }
    'confirm' { Show-Popup $Msg.id ([string]$Msg.url) ([int]$Msg.timeoutSec) ([int]$Msg.waiting) }
    'waiting' {
      if ($script:popup -and $script:popup.Id -eq $Msg.id) { $script:popup.WaitLabel.Text = Format-Waiting ([int]$Msg.count) }
    }
    'cancel' { if ($script:popup -and $script:popup.Id -eq $Msg.id) { Close-Popup $null } }
    'open' { Open-Url ([string]$Msg.url) }
    default { Send-Log ('unknown command: ' + $Msg.cmd) }
  }
}

function Stop-Helper {
  if ($script:stopping) { return }
  $script:stopping = $true
  $pump.Stop()
  Close-Popup $null
  $notify.Visible = $false
  $notify.Dispose()
  [System.Windows.Forms.Application]::Exit()
}

$pump = New-Object System.Windows.Forms.Timer
$pump.Interval = 50
$pump.Add_Tick({
  $line = $null
  while ([AiCenterStdin]::Lines.TryDequeue([ref]$line)) {
    if (-not $line.Trim()) { continue }
    try { Invoke-HelperCommand ($line | ConvertFrom-Json) }
    catch { Send-Log ('bad command: ' + $_.Exception.Message) }
  }
  if ([AiCenterStdin]::Closed -and [AiCenterStdin]::Lines.IsEmpty) { Stop-Helper }
  elseif ($script:exitDeadline -and (Get-Date) -gt $script:exitDeadline) { Stop-Helper }
})

Update-Tray
$notify.Visible = $true
[AiCenterStdin]::Start()
$pump.Start()
Send-Event @{ event = 'ready' }
[System.Windows.Forms.Application]::Run()
